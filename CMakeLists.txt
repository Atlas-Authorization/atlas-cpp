cmake_minimum_required(VERSION 3.16)
project(atlas-auth VERSION 0.1.0 LANGUAGES CXX)
add_library(atlas-auth INTERFACE)
add_library(atlas::atlas-auth ALIAS atlas-auth)
target_include_directories(atlas-auth INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(atlas-auth INTERFACE cxx_std_17)
install(DIRECTORY include/ DESTINATION include)
include(CTest)
if(BUILD_TESTING)
  add_executable(atlas_test tests/atlas_test.cpp)
  target_link_libraries(atlas_test atlas-auth)
  add_test(NAME atlas_test COMMAND atlas_test)
endif()
